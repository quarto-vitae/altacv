$-- `with` is the closer fit for the first line (an "@" icon suits an email
$-- address) and `why` for the second (an envelope icon suits a mailing
$-- address) -- the shape upstream's own sample gives its "Referees" section.

\cvref{$entries.what$}{$entries.with$}{$entries.why$}

$endfor$
